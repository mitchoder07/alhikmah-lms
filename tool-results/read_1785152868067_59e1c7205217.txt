   327→function FileUploader({ lessonId, onDone }: { lessonId: string; onDone: () => void }) {
   328→  const [uploading, setUploading] = useState(false)
   329→  const inputRef = useRef<HTMLInputElement>(null)
   330→
   331→  const upload = async (file: File) => {
   332→    if (file.size > 4 * 1024 * 1024) {
   333→      toast.error('File too large. Maximum 4MB.')
   334→      return
   335→    }
   336→    setUploading(true)
   337→    try {
   338→      const fd = new FormData()
   339→      fd.append('file', file)
   340→      const res = await fetch(`/api/admin/lessons/${lessonId}/files`, { method: 'POST', body: fd })
   341→      const text = await res.text()
   342→      if (!text) throw new Error('Upload failed. File may be too large (max 4MB).')
   343→      let json
   344→      try { json = JSON.parse(text) } catch { throw new Error('Server error during upload.') }
   345→      if (!res.ok) throw new Error(json.error || 'Upload failed')
   346→      toast.success(`Uploaded: ${file.name}`)
   347→      onDone()
   348→    } catch (e: any) {
   349→      toast.error(e.message)
   350→    } finally {
   351→      setUploading(false)
   352→      if (inputRef.current) inputRef.current.value = ''
   353→    }
   354→  }
   355→
   356→  return (
   357→    <>
   358→      <input
   359→        ref={inputRef}
   360→        type="file"
   361→        className="hidden"
   362→        accept="image/*,application/pdf,video/*,audio/*,.doc,.docx,.ppt,.pptx,.xls,.xlsx,.txt"
   363→        onChange={(e) => { const f = e.target.files?.[0]; if (f) upload(f) }}
   364→      />
   365→      <Button size="sm" variant="outline" disabled={uploading} onClick={() => inputRef.current?.click()}>
   366→        {uploading ? <Loader2 className="h-3 w-3 mr-1 animate-spin" /> : <Upload className="h-3 w-3 mr-1" />}
   367→        Upload File
   368→      </Button>
   369→    </>
   370→  )
   371→}

   373→function LessonEditor({ moduleId, lesson, courseId, onClose, onSaved }: any) {

   381→  })

   446→  const [passMark, setPassMark] = useState('50')
   447→  const [questions, setQuestions] = useState<Array<{ text: string; options: string[]; answer: number; imageUrl: string }>>([
   448→    { text: '', options: ['', '', '', ''], answer: 0, imageUrl: '' }
   449→  ])
   450→  const [saving, setSaving] = useState(false)
   451→  const [loading, setLoading] = useState(true)
   452→  const [existingQuizId, setExistingQuizId] = useState<string | null>(null)
   453→
   454→  // Fetch existing quiz for this lesson
   455→  useEffect(() => {
   456→    fetch(`/api/admin/quizzes?lessonId=${lessonId}`, { cache: 'no-store' })
   457→      .then(r => r.json())
   458→      .then(data => {
   459→        if (data.quiz) {
   460→          setExistingQuizId(data.quiz.id)
   461→          setTitle(data.quiz.title || '')
   462→          setPassMark(String(data.quiz.passMark || 50))
   463→          if (data.quiz.questions && data.quiz.questions.length > 0) {
   464→            setQuestions(data.quiz.questions.map((q: any) => ({
   465→              text: q.text,
   466→              options: q.options,
   467→              answer: Number(q.answer),
   468→              imageUrl: q.imageUrl || '',
   469→            })))
   470→          }
   471→        }
   472→      })
   473→      .catch(() => {})
   474→      .finally(() => setLoading(false))
   475→  }, [lessonId])
   476→
   477→  const updateQ = (i: number, field: 'text' | 'options' | 'answer' | 'imageUrl', val: any) => {
   478→    setQuestions(prev => prev.map((q, idx) => {
   479→      if (idx !== i) return q
   480→      if (field === 'options') {
   481→        const opts = [...q.options]
   482→        opts[val.idx] = val.value
   483→        return { ...q, options: opts }
   484→      }
   485→      return { ...q, [field]: val }
   486→    }))
   487→  }
   488→
   489→  const addQ = () => setQuestions([...questions, { text: '', options: ['', '', '', ''], answer: 0, imageUrl: '' }])
   490→  const insertQ = (i: number) => {
   491→    const next = [...questions]
   492→    next.splice(i + 1, 0, { text: '', options: ['', '', '', ''], answer: 0, imageUrl: '' })
   493→    setQuestions(next)
   494→  }
   495→  const removeQ = (i: number) => setQuestions(questions.filter((_, idx) => idx !== i))
   496→
   497→  const save = async () => {
   498→    if (!title || questions.some(q => !q.text || q.options.some(o => !o))) {
   499→      toast.error('Please fill all questions and options')
   500→      return
   501→    }
   502→    setSaving(true)
   503→    try {
   504→      await apiPost('/api/admin/quizzes', {
   505→        lessonId,
   506→        title,
   507→        passMark: Number(passMark),
   508→        questions: questions.map(q => ({
   509→          text: q.text,
   510→          options: q.options,
   511→          answer: q.answer,
   512→          imageUrl: q.imageUrl || null,
   513→        }))
   514→      })
   515→      toast.success(existingQuizId ? 'Quiz updated' : 'Quiz created')
   516→      onSaved()
   517→    } catch (e: any) { toast.error(e.message) } finally { setSaving(false) }
   518→  }
   519→
   520→  return (
   521→    <Dialog open onOpenChange={onClose}>
   522→      <DialogContent className="max-w-3xl max-h-[90vh] overflow-y-auto">
   523→        <DialogHeader>
   524→          <DialogTitle>{loading ? 'Loading...' : existingQuizId ? 'Edit Quiz' : 'Create Quiz'}</DialogTitle>
   525→          <DialogDescription>Add multiple-choice questions. The first correct option per question is the answer.</DialogDescription>
   526→        </DialogHeader>
   527→        <div className="space-y-4">
   528→          {loading ? (
   529→            <div className="flex items-center justify-center py-12">
   530→              <Loader2 className="h-6 w-6 animate-spin text-primary" />
   531→              <span className="ml-2 text-sm text-muted-foreground">Loading quiz...</span>
   532→            </div>
   533→          ) : (
   534→            <>
   535→          <div className="grid grid-cols-2 gap-3">
   536→            <div className="space-y-1.5">
   537→              <Label>Quiz Title</Label>
   538→              <Input value={title} onChange={(e) => setTitle(e.target.value)} placeholder="Lesson Quiz" />
   539→            </div>
   540→            <div className="space-y-1.5">
   541→              <Label>Pass Mark (%)</Label>
   542→              <Input type="number" value={passMark} onChange={(e) => setPassMark(e.target.value)} />
   543→            </div>
   544→          </div>
   545→
   546→          {questions.map((q, i) => (
   547→            <div key={i} className="border rounded-lg p-3 space-y-2">
   548→              <div className="flex items-center justify-between">
   549→                <Label className="text-xs">Question {i + 1}</Label>
   550→                {questions.length > 1 && <Button size="icon" variant="ghost" className="h-6 w-6" onClick={() => removeQ(i)}><X className="h-3 w-3" /></Button>}
   551→              </div>
   552→              <Input value={q.text} onChange={(e) => updateQ(i, 'text', e.target.value)} placeholder="Question text" />
   553→              <ImageUpload
   554→                value={q.imageUrl || null}
   555→                onChange={(url) => updateQ(i, 'imageUrl', url ?? '')}
   556→                label="Question Image (optional)"
   557→              />
   558→              <div className="space-y-1">
   559→                {q.options.map((opt, oi) => (
   560→                  <div key={oi} className="flex items-center gap-2">
   561→                    <input type="radio" name={`answer-${i}`} checked={q.answer === oi} onChange={() => updateQ(i, 'answer', oi)} className="h-3 w-3" />
   562→                    <Input value={opt} onChange={(e) => updateQ(i, 'options', { idx: oi, value: e.target.value })} placeholder={`Option ${oi + 1}`} className="text-xs h-8" />
   563→                  </div>
   564→                ))}
   565→              </div>
   566→              <p className="text-[10px] text-muted-foreground">Select the radio button next to the correct answer.</p>
   567→              <Button
   568→                variant="outline"
   569→                size="sm"
   570→                className="w-full text-xs"
   571→                onClick={() => insertQ(i)}
   572→              >
   573→                <Plus className="h-3 w-3 mr-1" /> Add Question Below
   574→              </Button>
   575→            </div>
   576→          ))}
   577→
   578→          <Button variant="outline" size="sm" onClick={addQ} className="w-full"><Plus className="h-3 w-3 mr-1" /> Add Question</Button>
   579→            </>
   580→          )}
   581→        </div>
   582→        <DialogFooter>
   583→          {existingQuizId && (
   584→            <Button variant="ghost" onClick={async () => {
   585→              if (!confirm('Delete this quiz? All student attempts will be kept but the quiz will no longer be available.')) return
   586→              try {
   587→                await apiDelete(`/api/admin/quizzes/${existingQuizId}`)
   588→                toast.success('Quiz deleted')
   589→                onSaved()
   590→              } catch (e: any) { toast.error(e.message) }
   591→            }} className="text-destructive mr-auto">
   592→              <Trash2 className="h-4 w-4 mr-1" /> Delete Quiz
   593→            </Button>
   594→          )}
   595→          <Button variant="outline" onClick={onClose}>Cancel</Button>
   596→          <Button onClick={save} disabled={saving || loading} className="bg-primary hover:bg-primary/90">
   597→            {saving ? <><Loader2 className="h-4 w-4 animate-spin mr-1" /> Saving...</> : (existingQuizId ? 'Update Quiz' : 'Create Quiz')}
   598→          </Button>
   599→        </DialogFooter>
   600→      </DialogContent>
   601→    </Dialog>
   602→  )
   603→}
